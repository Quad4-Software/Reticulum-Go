{{- $title := .Title -}} {{- if not $title -}} {{- with findRE `(?m)^#[ \t]+(\S.*?)[ \t#]*$` .RawContent -}} {{- $title = index . 0 | strings.TrimPrefix "#" | strings.TrimSpace | strings.TrimSuffix "#" | strings.TrimSpace -}} {{- end -}} {{- end -}} {{- if not $title -}} {{- with .File -}} {{- $title = .TranslationBaseName | humanize -}} {{- else -}} {{- $title = site.Title -}} {{- end -}} {{- end -}} {{- return $title -}}
